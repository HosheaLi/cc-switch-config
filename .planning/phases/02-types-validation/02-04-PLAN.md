    **AuthType Enum Schema:**
    ```typescript
    import { z } from 'zod';

    /**
     * Authentication type for API providers.
     */
    export const AuthTypeSchema = z.enum(['token', 'header', 'custom']);
    export type AuthType = z.infer<typeof AuthTypeSchema>;
    ```

    **ApiProviderConfigSchema:**
    ```typescript
    /**
     * API Provider Configuration Schema.
     * 
     * Defines a custom API provider with connection details.
     * Per D-09: Includes name, baseUrl, authType, headers.
     */
    export const ApiProviderConfigSchema = z.object({
      name: z.string().min(1, 'Provider name required'),
      baseUrl: z.string().url('Valid URL required'),
      authType: AuthTypeSchema,
      headers: z.record(z.string(), z.string()).optional(),
      // Additional env vars to set when using this provider
      env: z.record(z.string(), z.string()).optional(),
    }).strict();

    export type ApiProviderConfig = z.infer<typeof ApiProviderConfigSchema>;
    ```

    **TemplateConfigSchema:**
    ```typescript
    /**
     * Template Configuration Schema.
     * 
     * Defines a reusable provider template.
     * Per F7: User-defined templates, no predefined templates.
     */
    export const TemplateConfigSchema = z.object({
      name: z.string().min(1, 'Template name required'),
      description: z.string().optional(),
      provider: ApiProviderConfigSchema,
      // Tags for organization/filtering
      tags: z.array(z.string()).optional(),
      // Created/updated timestamps
      createdAt: z.string().datetime().optional(),
      updatedAt: z.string().datetime().optional(),
    }).strict();

    export type TemplateConfig = z.infer<typeof TemplateConfigSchema>;
    ```

    **TemplateStoreSchema (for template management):**
    ```typescript
    /**
     * Template Store Schema.
     * 
     * Schema for templates.json storage file.
     */
    export const TemplateStoreSchema = z.object({
      version: z.number().int().optional(),
      templates: z.record(z.string(), TemplateConfigSchema),
    }).strict();

    export type TemplateStore = z.infer<typeof TemplateStoreSchema>;
    ```

    **Implementation Notes:**
    - AuthType enum: 'token' (env token), 'header' (auth header), 'custom' (flexible)
    - ApiProviderConfig.baseUrl must be valid URL format
    - TemplateConfig.name is the unique identifier (used as key in TemplateStore)
    - All schemas use `.strict()` to reject unknown fields
    - JSDoc comments for each schema explaining purpose

    import { describe, it, expect } from 'vitest';
    import {
      ApiProviderConfigSchema,
      TemplateConfigSchema,
      AuthTypeSchema,
    } from './provider.js';

    describe('AuthTypeSchema', () => {
      it('accepts valid auth types', () => {
        expect(AuthTypeSchema.parse('token')).toBe('token');
        expect(AuthTypeSchema.parse('header')).toBe('header');
        expect(AuthTypeSchema.parse('custom')).toBe('custom');
      });

      it('rejects invalid auth type', () => {
        const result = AuthTypeSchema.safeParse('oauth');
        expect(result.success).toBe(false);
      });
    });

    describe('ApiProviderConfigSchema', () => {
      it('accepts valid provider config', () => {
        const config = {
          name: 'OpenRouter',
          baseUrl: 'https://openrouter.ai/api/v1',
          authType: 'header',
          headers: { 'Authorization': 'Bearer key' }
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(true);
      });

      it('rejects invalid URL', () => {
        const config = {
          name: 'BadProvider',
          baseUrl: 'not-a-url',
          authType: 'token'
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(false);
      });

      it('rejects missing name', () => {
        const config = {
          baseUrl: 'https://api.example.com',
          authType: 'token'
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(false);
      });

      it('rejects invalid authType', () => {
        const config = {
          name: 'Test',
          baseUrl: 'https://api.example.com',
          authType: 'invalid'
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(false);
      });

      it('accepts optional headers and env', () => {
        const config = {
          name: 'Provider',
          baseUrl: 'https://api.example.com',
          authType: 'token',
          headers: { 'X-Custom': 'value' },
          env: { CUSTOM_VAR: 'value' }
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(true);
      });

      it('rejects unknown keys', () => {
        const config = {
          name: 'Provider',
          baseUrl: 'https://api.example.com',
          authType: 'token',
          unknownField: 'value'
        };
        const result = ApiProviderConfigSchema.safeParse(config);
        expect(result.success).toBe(false);
      });
    });

    describe('TemplateConfigSchema', () => {
      it('accepts valid template', () => {
        const template = {
          name: 'my-openrouter',
          description: 'OpenRouter API template',
          provider: {
            name: 'OpenRouter',
            baseUrl: 'https://openrouter.ai/api/v1',
            authType: 'header'
          }
        };
        const result = TemplateConfigSchema.safeParse(template);
        expect(result.success).toBe(true);
      });

      it('rejects template without provider', () => {
        const template = {
          name: 'broken-template',
          description: 'Missing provider'
        };
        const result = TemplateConfigSchema.safeParse(template);
        expect(result.success).toBe(false);
      });

      it('validates nested provider schema', () => {
        const template = {
          name: 'template',
          provider: {
            name: 'Bad',
            baseUrl: 'invalid-url', // Invalid URL in nested provider
            authType: 'token'
          }
        };
        const result = TemplateConfigSchema.safeParse(template);
        expect(result.success).toBe(false);
      });

      it('accepts optional tags and timestamps', () => {
        const template = {
          name: 'template',
          provider: {
            name: 'Provider',
            baseUrl: 'https://api.example.com',
            authType: 'token'
          },
          tags: ['ai', 'llm'],
          createdAt: '2026-04-13T10:00:00Z',
          updatedAt: '2026-04-13T11:00:00Z'
        };
        const result = TemplateConfigSchema.safeParse(template);
        expect(result.success).toBe(true);
      });

      it('rejects unknown keys', () => {
        const template = {
          name: 'template',
          provider: {
            name: 'Provider',
            baseUrl: 'https://api.example.com',
            authType: 'token'
          },
          unknownField: 'value'
        };
        const result = TemplateConfigSchema.safeParse(template);
        expect(result.success).toBe(false);
      });
    });
