    Implementation:
    ```typescript
    import { useInput, Key } from 'ink';

    interface KeyInputOptions {
      onUp?: () => void;
      onDown?: () => void;
      onSelect?: () => void;
      onEscape?: () => void;
      isActive?: boolean;
    }

    export function useKeyInput(options: KeyInputOptions) {
      useInput((input, key) => {
        if (options.isActive === false) return;

        // Arrow keys
        if (key.upArrow) options.onUp?.();
        if (key.downArrow) options.onDown?.();

        // Vim-style j/k (U3 requirement)
        if (input === 'k') options.onUp?.();
        if (input === 'j') options.onDown?.();

        // Escape (U4 requirement)
        if (key.escape) options.onEscape?.();

        // Enter for selection
        if (key.return) options.onSelect?.();
      }, { isActive: options.isActive ?? true });
    }
    ```

    ```typescript
    import { useState } from 'react';

    export type Screen = 'list' | 'editor' | 'confirm' | 'template-select';

    interface NavigationState {
      stack: Screen[];
      current: Screen;
      isRoot: boolean;
    }

    interface NavigationActions {
      push: (screen: Screen) => void;
      pop: () => void;
      reset: () => void;
    }

    export function useNavigation(initialScreen: Screen = 'list'): NavigationState & NavigationActions {
      const [stack, setStack] = useState<Screen[]>([initialScreen]);

      const push = (screen: Screen) => {
        setStack(prev => [...prev, screen]);
      };

      const pop = () => {
        setStack(prev => prev.length > 1 ? prev.slice(0, -1) : prev);
      };

      const reset = () => {
        setStack([initialScreen]);
      };

      return {
        stack,
        current: stack[stack.length - 1],
        isRoot: stack.length === 1,
        push,
        pop,
        reset,
      };
    }
    ```

    ```typescript
    import Fuse, { IFuseOptions } from 'fuse.js';
    import { useMemo } from 'react';

    export interface SearchableItem {
      name: string;
      path: string;
    }

    export function useFuzzySearch<T extends SearchableItem>(
      items: T[],
      query: string,
      options?: Partial<IFuseOptions<T>>
    ): T[] {
      const defaultOptions: IFuseOptions<T> = {
        keys: ['name', 'path'],
        threshold: 0.4,        // Balance precision vs recall (D-06)
        includeMatches: true, // For highlighting
        ignoreLocation: true, // Better for long strings
        ...options,
      };

      const fuse = useMemo(() => new Fuse(items, defaultOptions), [items, defaultOptions]);

      return useMemo(() => {
        if (!query.trim()) return items;
        return fuse.search(query).map(result => result.item);
      }, [fuse, query, items]);
    }
    ```
