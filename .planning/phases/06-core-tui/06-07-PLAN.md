    ```typescript
     * TUI Launch Utility - Real Implementation (Phase 06)

     * Per D-02: No args launches TUI.

     */

    import { runTUI } from '../../tui/index.js';
    import { TemplateService } from '../../lib/services/index.js';
    import { TemplateStore } from '../../lib/store/index.js';
    import { readConfig, writeConfig } from '../../lib/store/config.js';
    import chalk from 'chalk';

    /**
     * Launch the main TUI application.
     * Per D-02: Called when no CLI arguments provided.
     */
    export async function launchTUI(): Promise<void> {
      await runTUI();
    }

    /**
     * Launch TUI for template selection.
     * Per D-06: Called when switch command has no argument.
     *
     * @returns Selected template name or null if cancelled
     */
    export async function selectTemplateInTUI(): Promise<string | null> {
      // Create template service for listing templates
      const templateStore = new TemplateStore();
      const templateService = new TemplateService(templateStore, readConfig, writeConfig);

      const templates = await templateService.listTemplates();

      if (templates.length === 0) {
        console.log(chalk.yellow('No templates available.'));
        console.log(chalk.gray('Create a template first: cc-config template create <name>'));
        return null;
      }

      // For now, use a simple selection approach
      // Full TUI template selection would require a TemplateSelectScreen
      // This is a placeholder that lists templates for user to type

      console.log(chalk.cyan('Available templates:'));
      for (const name of templates) {
        console.log(chalk.white(`  - ${name}`));
      }
      console.log(chalk.gray('\nPlease specify template name: cc-config switch <template-name>'));

      return null;  // Return null to indicate user should use CLI with argument
    }
    ```

    Note: selectTemplateInTUI is simplified for Phase 06.

    ```typescript
    import { describe, it, expect } from 'vitest';
    import fs from 'fs';
    import path from 'path';

    describe('M4 Architectural Boundary', () => {
      const servicesDir = path.join(process.cwd(), 'src/lib/services');
      const tuiDir = path.join(process.cwd(), 'src/tui');
      const cliDir = path.join(process.cwd(), 'src/cli');

      it('Services should NOT import ink', () => {
        const files = fs.readdirSync(servicesDir).filter(f => f.endsWith('.ts'));
        for (const file of files) {
          const content = fs.readFileSync(path.join(servicesDir, file), 'utf-8');
          expect(content).not.toMatch(/from ['"]ink['"]/);
          expect(content).not.toMatch(/from ['"]react['"]/);
        }
      });

      it('Services should NOT import from TUI', () => {
        const files = fs.readdirSync(servicesDir).filter(f => f.endsWith('.ts'));
        for (const file of files) {
          const content = fs.readFileSync(path.join(servicesDir, file), 'utf-8');
          expect(content).not.toMatch(/from ['"].*tui['"]/);
        }
      });

      it('TUI should import from Services (correct dependency direction)', () => {
        // TUI can import Services - this is correct
        const appContent = fs.readFileSync(path.join(tuiDir, 'app.tsx'), 'utf-8');
        expect(appContent).toMatch(/from ['"].*services['"]/);

        const cliFiles = fs.readdirSync(cliDir).filter(f => f.endsWith('.ts') && !f.includes('tui-launch'));
    ```

    ```typescript
    function getAllFiles(dir: string): string[] {
      const files: string[] = [];
      for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
        if (entry.isDirectory()) {
          files.push(...getAllFiles(path.join(dir, entry.name)));
        } else if (entry.name.endsWith('.ts') || entry.name.endsWith('.tsx')) {
          files.push(path.join(dir, entry.name));
        }
      }
      return files;
    }
    ```

    ```typescript
    import { describe, it, expect } from 'vitest';
    import { render } from 'ink-testing-library';
    import React from 'react';
    import { ProjectListScreen } from './screens/ProjectListScreen.js';
    import { useFuzzySearch } from './hooks/useFuzzySearch.js';
    import type { ProjectEntry } from '../lib/store/project.js';

    describe('N4 Performance Tests', () => {
      // Generate 100 mock projects
      const generateMockProjects = (count: number): ProjectEntry[] => {
        return Array.from({ length: count }, (_, i) => ({
          id: `project-${i}`,
          path: `/Users/test/projects/project-${i}`,
          activeConfig: i % 3 === 0 ? `template-${i % 10}` : null,
          lastModified: new Date().toISOString(),
        }));
      };

      it('should render 100 projects in <50ms', () => {
        const projects = generateMockProjects(100);
        const start = performance.now();

        const { lastFrame } = render(
          <ProjectListScreen
            projects={projects}
            onSelect={() => {}}
            onExit={() => {}}
          />
        );

        const renderTime = performance.now() - start;

        // Verify render completed
        expect(lastFrame()).toContain('Projects');

        // N4 requirement: <50ms
        expect(renderTime).toBeLessThan(50);
      });

      it('should fuzzy search 100 items in <10ms', () => {
        const items = generateMockProjects(100).map(p => ({
          ...p,
          name: p.path.split('/').pop() ?? p.path,
        }));

        const start = performance.now();

        // Simulate useFuzzySearch behavior
        const Fuse = require('fuse.js').default;
        const fuse = new Fuse(items, {
          keys: ['name', 'path'],
          threshold: 0.4,
        });
        const results = fuse.search('project-5');

        const searchTime = performance.now() - start;

        expect(results.length).toBeGreaterThan(0);
        expect(searchTime).toBeLessThan(10);
      });
    });
    ```
