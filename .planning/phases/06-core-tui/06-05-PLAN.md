    Create ConfirmScreen with explicit y/n confirmation.

    Implementation:
    ```typescript
    import React from 'react';
    import { Box, Text, useInput } from 'ink';
    import { useNavigation } from '../hooks/useNavigation.js';

    interface ConfirmScreenProps {
      message: string;
      actionDescription: string;
      onConfirm: () => void;
      onCancel: () => void;
    }

    export const ConfirmScreen: React.FC<ConfirmScreenProps> = ({
      message,
      actionDescription,
      onConfirm,
      onCancel,
    }) => {
      // Custom useInput for y/n handling (not useKeyInput since we need specific letter input)
      useInput((input, key) => {
        if (input.toLowerCase() === 'y') {
          onConfirm();
        }
        if (input.toLowerCase() === 'n' || key.escape) {
          onCancel();
        }
        // Enter is ignored - user must explicitly type y/n
      });

      return (
        <Box flexDirection="column" padding={2} justifyContent="center" alignItems="center">
          {/* Warning Icon */}
          <Text bold color="red">
            ⚠ WARNING
          </Text>

          {/* Confirmation Message */}
          <Box marginTop={1}>
            <Text bold>{message}</Text>
          </Box>

          {/* Action Description */}
          <Box marginTop={1} borderStyle="single" borderColor="red" padding={1}>
            <Text dimColor>{actionDescription}</Text>
          </Box>

          {/* Prompt */}
          <Box marginTop={2}>
            <Text bold color="yellow">
              Type 'y' to confirm, 'n' to cancel (or Esc)
            </Text>
          </Box>
        </Box>
      );
    };
    ```

    Note: Uses direct useInput for y/n handling, not useKeyInput, since we need specific letter input.
    Enter is deliberately ignored to force explicit y/n confirmation (U5).

    Create test file verifying y/n handling, escape cancellation, and message display.
