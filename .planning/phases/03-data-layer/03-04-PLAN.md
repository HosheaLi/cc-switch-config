interface FSWatcher {

  add(paths: string | string[]): void;
  unwatch(paths: string | string[]): void;

chokidar.watch(paths: string | string[], options?: WatchOptions): FSWatcher;

interface WatchOptions {
  ignoreInitial?: boolean; // Don't fire on initial scan (default: true)
  awaitWriteFinish?: {
    stabilityThreshold: number; // Debounce time in ms
    pollInterval: number; // Polling interval
  };
}
