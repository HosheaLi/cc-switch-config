    **Step 2: Implement formatters.ts:**

    ```typescript
    /**
     * Text Formatting Functions Module
     *
     * Per UI-SPEC.md Copywriting Contract:
     * - Message: accent (blue) for prompts
     * - Hint: muted (gray) for descriptions
     * - Error: danger (red) with ✗ symbol
     * - Success: success (green) with ✓ symbol
     * - Warning: warning (yellow) with ⚠ symbol
     * - Separator: horizontal line (Unicode or ASCII)
     */

    import { colors } from './colors.js';
    import { getBorders } from './borders.js';

    /**
     * Format a prompt message with accent color.
     * Per UI-SPEC: Used for interactive prompts, primary actions.
     *
     * @param text - Message text
     * @returns Formatted message
     */
    export function message(text: string): string {
      return colors.accent(text);
    }

    /**
     * Format a hint/description with muted color.
     * Per UI-SPEC: Used for secondary information, guidance.
     *
     * @param text - Hint text
     * @returns Formatted hint
     */
    export function hint(text: string): string {
      return colors.muted(text);
    }

    /**
     * Format an error message with danger color and ✗ symbol.
     * Per UI-SPEC: "✗ [Validation error message]"
     *
     * @param text - Error text
     * @returns Formatted error with symbol
     */
    export function error(text: string): string {
      return colors.danger(`✗ ${text}`);
    }

    /**
     * Format a success message with success color and ✓ symbol.
     * Per UI-SPEC: "✓ Configuration applied successfully"
     *
     * @param text - Success text
     * @returns Formatted success with symbol
     */
    export function success(text: string): string {
      return colors.success(`✓ ${text}`);
    }

    /**
     * Format a warning message with warning color and ⚠ symbol.
     * Per UI-SPEC: "⚠ API key will be replaced"
     *
     * @param text - Warning text
     * @returns Formatted warning with symbol
     */
    export function warning(text: string): string {
      return colors.warning(`⚠ ${text}`);
    }

    /**
     * Create a separator line for visual grouping.
     * Per UI-SPEC: 40 characters width by default.
     *
     * @param width - Line width (default: 40)
     * @returns Horizontal separator line
     */
    export function separator(width: number = 40): string {
      const borders = getBorders();
      return colors.muted(borders.horizontal.repeat(width));
    }

    /**
     * Format a cancel option with danger color.
     * Per UI-SPEC: Used for "取消" options in prompts.
     *
     * @param text - Cancel text (default: "取消")
     * @returns Formatted cancel option
     */
    export function cancel(text: string = '取消'): string {
      return colors.danger(text);
    }
    ```
