    /**
     * Terminal Capability Detection Module
     *
     * Per D-08, D-09: NO_COLOR completely disables all ANSI colors.
     * Per D-10: TERM_PROGRAM for macOS terminals, WT_SESSION for Windows Terminal.
     * Per D-11: Windows CMD has ANSI support (Win 10+) but no truecolor.
     */

    export interface ColorSupport {
      /** ANSI colors supported */
      enabled: boolean;
      /** 24-bit truecolor (RGB) supported */
      truecolor: boolean;
      /** Detection reason for debugging */
      reason: string;
    }

    /**
     * Detect terminal color support capabilities.
     *
     * Detection order (per RESEARCH.md):
     * 1. NO_COLOR → disable all colors (D-08, D-09)
     * 2. FORCE_COLOR → override, enable colors
     * 3. WT_SESSION → Windows Terminal, truecolor supported (D-10)
     * 4. COLORTERM → truecolor indicator
     * 5. TERM_PROGRAM → macOS terminals (iTerm2, Apple Terminal)
     * 6. Windows platform → ANSI support, no truecolor (D-11)
     * 7. Default → basic ANSI, no truecolor
     */
    export function detectColorSupport(): ColorSupport {
      const env = process.env;

      // D-08, D-09: NO_COLOR completely disables colors
      if (env.NO_COLOR && env.NO_COLOR !== '') {
        return { enabled: false, truecolor: false, reason: 'NO_COLOR set' };
      }

      // FORCE_COLOR overrides detection
      if (env.FORCE_COLOR) {
        return { enabled: true, truecolor: false, reason: 'FORCE_COLOR set' };
      }

      // D-10: Windows Terminal (WT_SESSION)
      if (env.WT_SESSION) {
        return { enabled: true, truecolor: true, reason: 'Windows Terminal' };
      }

      // COLORTERM indicates truecolor support
      if (env.COLORTERM === 'truecolor' || env.COLORTERM === '24bit') {
        return { enabled: true, truecolor: true, reason: 'COLORTERM=truecolor' };
      }

      // D-10: macOS terminals (TERM_PROGRAM)
      if (env.TERM_PROGRAM === 'iTerm.app') {
        return { enabled: true, truecolor: true, reason: 'iTerm2' };
      }
      if (env.TERM_PROGRAM === 'Apple_Terminal') {
        return { enabled: true, truecolor: false, reason: 'macOS Terminal' };
      }
      if (env.TERM_PROGRAM === 'vscode') {
        return { enabled: true, truecolor: true, reason: 'VSCode terminal' };
      }

      // D-11: Windows CMD (ANSI support, no truecolor)
      if (process.platform === 'win32') {
        return { enabled: true, truecolor: false, reason: 'Windows ANSI' };
      }

      // Default: basic ANSI support
      return { enabled: true, truecolor: false, reason: 'default' };
    }

    /**
     * Singleton color support detection result.
     * Computed once at module load for consistent behavior.
     */
    export const colorSupport = detectColorSupport();

    **Step 2: Implement colors.ts:**

    ```typescript
    /**
     * Color Definitions Module
     *
     * Per D-01: Use picocolors for standard ANSI colors.
     * Per D-02: OpenCode warm palette via custom truecolor ANSI codes.
     * Per D-03: Apple HIG semantic colors (blue/red/green/yellow).
     */

    import pc from 'picocolors';
    import { colorSupport } from './detection.js';
    import type { ColorSupport } from './detection.js';

    /**
     * OpenCode warm palette ANSI codes (per D-02).
     *
     * #201d1d → RGB(32, 29, 29) → dark background
     * #fdfcfc → RGB(253, 252, 252) → light foreground
     * #9a9898 → RGB(154, 152, 152) → muted text
     */
    export const OPENCODE_PALETTE = {
      /** Background: #201d1d */
      darkBg:  '\\x1b[48;2;32;29;29m',
      /** Foreground: #fdfcfc */
      lightFg: '\\x1b[38;2;253;252;252m',
      /** Muted: #9a9898 */
      muted:   '\\x1b[38;2;154;152;152m',
      /** Reset all ANSI codes */
      reset:   '\\x1b[0m',
    };

    /**
     * Create color functions based on terminal capabilities.
     *
     * @param support - Color support detection result
     * @returns Color function object
     */
    export function createColors(support: ColorSupport) {
      // Formatter for truecolor: returns (text) => colored text or plain text
      const truecolorFg = (code: string) =>
        support.enabled && support.truecolor
          ? (text: string) => `${code}${text}\\x1b[39m`
          : (text: string) => text;

      // For non-truecolor terminals, use picocolors gray for muted
      const mutedFormatter = support.truecolor
        ? truecolorFg(OPENCODE_PALETTE.muted)
        : support.enabled ? pc.gray : (s: string) => s;

      return {
        // Semantic colors (per D-03) - picocolors standard
        accent:   support.enabled ? pc.blue   : (s: string) => s,
        danger:   support.enabled ? pc.red    : (s: string) => s,
        success:  support.enabled ? pc.green  : (s: string) => s,
        warning:  support.enabled ? pc.yellow : (s: string) => s,

        // OpenCode palette (per D-02) - custom truecolor
        background: support.enabled && support.truecolor
          ? (text: string) => `${OPENCODE_PALETTE.darkBg}${text}${OPENCODE_PALETTE.reset}`
          : (text: string) => text,
        foreground: truecolorFg(OPENCODE_PALETTE.lightFg),
        muted: mutedFormatter,

        // Modifiers - picocolors
        bold:     support.enabled ? pc.bold    : (s: string) => s,
        dim:      support.enabled ? pc.dim     : (s: string) => s,
        italic:   support.enabled ? pc.italic  : (s: string) => s,
        underline: support.enabled ? pc.underline : (s: string) => s,

        // Capability flag
        isColorSupported: support.enabled,
        isTruecolor: support.truecolor,
      };
    }

    /**
     * Singleton colors object using detected terminal capabilities.
     */
    export const colors = createColors(colorSupport);
    ```

    /**
     * Border Characters Module
     *
     * Per D-06: Flat depth system, border-only elevation.
     * Per D-07: Single-line borders, no gradients/embossing.
     * Per D-11: ASCII fallback for Windows CMD.
     */

    /**
     * Border character definitions.
     *
     * Unicode: Modern terminals (Windows Terminal, iTerm2, VSCode, truecolor terminals)
     * ASCII: Windows CMD and basic terminals
     */
    export const BORDERS = {
      /** Unicode box-drawing characters (U+2500 block) */
      unicode: {
        horizontal: '─',  // U+2500
        vertical:   '│',  // U+2502
        topLeft:    '┌',  // U+250C
        topRight:   '┐',  // U+2510
        bottomLeft: '└',  // U+2514
        bottomRight: '┘', // U+2518
        cross:      '┼',  // U+253C
      },

      /** ASCII fallback (per D-11: Windows CMD compatibility) */
      ascii: {
        horizontal: '-',
        vertical:   '|',
        topLeft:    '+',
        topRight:   '+',
        bottomLeft: '+',
        bottomRight: '+',
        cross:      '+',
      },
    };

    /**
     * Get appropriate border characters based on terminal capabilities.
     *
     * Selection logic (per UI-SPEC.md):
     * - Unicode: Windows Terminal, iTerm2, VSCode, COLORTERM=truecolor
     * - ASCII: Windows CMD (win32 without WT_SESSION)
     *
     * @returns Border character set
     */
    export function getBorders() {
      const env = process.env;

      // Windows Terminal: Unicode supported
